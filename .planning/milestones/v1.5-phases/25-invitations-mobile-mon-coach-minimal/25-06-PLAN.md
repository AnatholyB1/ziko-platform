import { describe, it, expect, beforeAll, afterAll, vi } from 'vitest';
import { getAdminClient, createTestUser, cleanupTestUsers, type TestUser } from '../rls/fixtures';
import { insertInvitation, listInvitations, revokeInvitation } from '../../src/coach/invitations/db.js';

const adminClient = getAdminClient();

let coach: TestUser;
const createdInvitationIds: string[] = [];

beforeAll(async () => {
  coach = await createTestUser('inv-coach');
  await adminClient.from('user_profiles').upsert({ id: coach.id, role: 'coach', display_name: 'TestCoach' });
  await adminClient.from('coach_profiles').upsert({ user_id: coach.id, display_name: 'TestCoach' });
});

afterAll(async () => {
  if (createdInvitationIds.length) {
    await adminClient.from('coach_invitations').delete().in('id', createdInvitationIds);
  }
  await adminClient.from('coach_invitations').delete().eq('coach_id', coach.id);
  await adminClient.from('coach_profiles').delete().eq('user_id', coach.id);
  await cleanupTestUsers([coach.id]);
});

describe('coach/invitations/db.insertInvitation (INVITE-01)', () => {
  it('inserts row with 6-char [A-Z2-9] code matching DB CHECK', async () => {
    const { data: s } = await coach.client.auth.getSession();
    const jwt = s.session!.access_token;
    const row = await insertInvitation(jwt, coach.id, { expires_at: null });
    createdInvitationIds.push(row.id);
    expect(row.code).toMatch(/^[A-Z2-9]{6}$/);
    expect(row.coach_id).toBe(coach.id);
    expect(row.expires_at).toBeNull();
    expect(row.use_count).toBe(0);
    expect(row.max_uses).toBeGreaterThanOrEqual(1);
  });

  it('respects 14-day default expiration when caller provides ISOString', async () => {
    const { data: s } = await coach.client.auth.getSession();
    const jwt = s.session!.access_token;
    const expires = new Date(Date.now() + 14 * 24 * 60 * 60 * 1000).toISOString();
    const row = await insertInvitation(jwt, coach.id, { expires_at: expires });
    createdInvitationIds.push(row.id);
    expect(row.expires_at).not.toBeNull();
    const diff = Math.abs(new Date(row.expires_at!).getTime() - new Date(expires).getTime());
    expect(diff).toBeLessThan(2000);
  });

  it('retries up to 3 times on PG 23505 unique violation (deterministic mock)', async () => {
    // Deterministic mock: 2 dupes then unique
    const mockCodes = ['DUPLI1', 'DUPLI1', 'UNIQUE'];
    let callIdx = 0;
    vi.doMock('nanoid', () => ({
      customAlphabet: () => () => mockCodes[callIdx++],
    }));
    vi.resetModules();
    const { insertInvitation: retryInsert } = await import('../../src/coach/invitations/db.js');

    // Pre-insert 'DUPLI1' to force collision on first two attempts
    const pre = await adminClient
      .from('coach_invitations')
      .insert({
        coach_id: coach.id,
        code: 'DUPLI1',
        expires_at: new Date(Date.now() + 7 * 86400000).toISOString(),
      })
      .select()
      .single();
    if (pre.data) createdInvitationIds.push(pre.data.id);

    // Now retryInsert should encounter 23505 twice (DUPLI1 already exists), then succeed with UNIQUE on the 3rd attempt
    const { data: s } = await coach.client.auth.getSession();
    const jwt = s.session!.access_token;
    const result = await retryInsert(jwt, coach.id, { expires_at: null });
    if (result?.id) createdInvitationIds.push(result.id);

    expect(result.code).toBe('UNIQUE');
    expect(callIdx).toBe(3); // confirms exactly 3 attempts were made

    vi.resetModules();
    vi.unmock('nanoid');
  });

  it('throws on non-23505 db errors immediately (no retry)', async () => {
    const { data: s } = await coach.client.auth.getSession();
    const jwt = s.session!.access_token;
    // FK violation 23503 — invalid coach_id
    await expect(insertInvitation(jwt, '00000000-0000-0000-0000-000000000000', { expires_at: null }))
      .rejects.toThrow();
  });
});

describe('coach/invitations/db.listInvitations (INVITE-02)', () => {
  it('returns rows with computed status field', async () => {
    const { data: s } = await coach.client.auth.getSession();
    const jwt = s.session!.access_token;
    const rows = await listInvitations(jwt, coach.id, 'all');
    expect(Array.isArray(rows)).toBe(true);
    expect(rows.length).toBeGreaterThan(0);
    for (const r of rows) expect(['active','used','expired','revoked']).toContain(r.status);
  });

  it('filter=active excludes expired and revoked rows', async () => {
    const expired = await adminClient.from('coach_invitations').insert({
      coach_id: coach.id,
      code: 'EXP123',
      expires_at: new Date(Date.now() - 1000).toISOString(),
    }).select().single();
    if (expired.data) createdInvitationIds.push(expired.data.id);

    const { data: s } = await coach.client.auth.getSession();
    const jwt = s.session!.access_token;
    const active = await listInvitations(jwt, coach.id, 'active');
    expect(active.every(r => r.status === 'active')).toBe(true);
    expect(active.find(r => r.id === expired.data!.id)).toBeUndefined();
  });

  it('orders by created_at DESC', async () => {
    const { data: s } = await coach.client.auth.getSession();
    const jwt = s.session!.access_token;
    const rows = await listInvitations(jwt, coach.id, 'all');
    for (let i = 1; i < rows.length; i++) {
      expect(new Date(rows[i-1].created_at).getTime()).toBeGreaterThanOrEqual(new Date(rows[i].created_at).getTime());
    }
  });
});

describe('coach/invitations/db.revokeInvitation (INVITE-02)', () => {
  it('sets revoked_at on active row owned by caller', async () => {
    const { data: s } = await coach.client.auth.getSession();
    const jwt = s.session!.access_token;
    const row = await insertInvitation(jwt, coach.id, { expires_at: null });
    createdInvitationIds.push(row.id);
    const r = await revokeInvitation(jwt, coach.id, row.id);
    expect(r.revoked_at).not.toBeNull();
    const { data } = await adminClient.from('coach_invitations').select('revoked_at').eq('id', row.id).single();
    expect(data!.revoked_at).not.toBeNull();
  });

  it('is idempotent — second call returns success without throwing', async () => {
    const { data: s } = await coach.client.auth.getSession();
    const jwt = s.session!.access_token;
    const row = await insertInvitation(jwt, coach.id, { expires_at: null });
    createdInvitationIds.push(row.id);
    await revokeInvitation(jwt, coach.id, row.id);
    await expect(revokeInvitation(jwt, coach.id, row.id)).resolves.toBeDefined();
  });
});

import { describe, it, expect, beforeAll, afterAll } from 'vitest';
import { getAdminClient, createTestUser, cleanupTestUsers, type TestUser } from '../rls/fixtures';
import { peekInvitation } from '../../src/coach/clients/db.js';

const adminClient = getAdminClient();

let coach: TestUser;
let client: TestUser;
let clientJwt: string;
const codes: Record<string, string> = {};
const ids: string[] = [];

beforeAll(async () => {
  coach = await createTestUser('cp-coach');
  client = await createTestUser('cp-client');
  await adminClient.from('user_profiles').upsert({ id: coach.id, role: 'coach', display_name: 'Coach' });
  await adminClient.from('coach_profiles').upsert({
    user_id: coach.id,
    display_name: 'Coach Name',
    bio: 'Hi',
    specialties: ['Hyrox'],
    kyc_status: 'verified',
  });
  await adminClient.from('user_profiles').upsert({ id: client.id, role: 'client' });

  const { data: s } = await client.client.auth.getSession();
  clientJwt = s.session!.access_token;

  async function mk(over: Partial<{ revoked_at: string; expires_at: string; use_count: number; max_uses: number }>) {
    const code = Math.random().toString(36).toUpperCase().replace(/[^A-Z2-9]/g, '').slice(0, 6).padEnd(6, '2');
    const { data } = await adminClient.from('coach_invitations').insert({ coach_id: coach.id, code, ...over }).select().single();
    ids.push(data!.id);
    return { code, id: data!.id };
  }
  codes.valid = (await mk({})).code;
  codes.revoked = (await mk({ revoked_at: new Date().toISOString() })).code;
  codes.expired = (await mk({ expires_at: new Date(Date.now() - 1000).toISOString() })).code;
  codes.usedUp = (await mk({ use_count: 1, max_uses: 1 })).code;
  // LINK_EXISTS: pre-create a link so client already has active link for THIS coach
  const linkUp = await adminClient.from('coach_client_links').insert({ coach_id: coach.id, client_id: client.id, invitation_id: ids[0] }).select().single();
  if (linkUp.data) ids.push(linkUp.data.id);
  codes.linkExists = (await mk({})).code;
});

afterAll(async () => {
  await adminClient.from('coach_client_links').delete().eq('client_id', client.id);
  if (ids.length) await adminClient.from('coach_invitations').delete().eq('coach_id', coach.id);
  await adminClient.from('coach_profiles').delete().eq('user_id', coach.id);
  await cleanupTestUsers([coach.id, client.id]);
});

describe('coach/clients/db.peekInvitation (INVITE-05)', () => {
  it('returns ok=true + preview on valid code (after clearing pre-existing link first)', async () => {
    // Isolate by using a new client with no link
    const fresh = await createTestUser('cp-fresh');
    await adminClient.from('user_profiles').upsert({ id: fresh.id, role: 'client' });
    const { data: s } = await fresh.client.auth.getSession();
    const result = await peekInvitation(s.session!.access_token, { code: codes.valid });
    expect(result.ok).toBe(true);
    if (result.ok) {
      expect(result.preview.display_name).toBe('Coach Name');
      expect(result.preview.specialties).toContain('Hyrox');
      expect(result.preview.kyc_status).toBe('verified');
    }
    await cleanupTestUsers([fresh.id]);
  });
});

describe('coach/clients/db.peekInvitation error collapsing (INVITE-07 + T-25-01)', () => {
  it.each([
    ['nonexistent code → INVALID_OR_EXPIRED', 'ZZZ999'],
    ['revoked code', codes.revoked],
    ['expired code', codes.expired],
    ['already-used code', codes.usedUp],
    ['link already exists', codes.linkExists],
  ])('%s', async (_label, code) => {
    const res = await peekInvitation(clientJwt, { code });
    expect(res).toEqual({ ok: false, error_code: 'INVALID_OR_EXPIRED', preview: null });
  });

  it('all error envelopes are byte-identical (deep-equal)', async () => {
    const a = await peekInvitation(clientJwt, { code: 'ZZZ999' });
    const b = await peekInvitation(clientJwt, { code: codes.expired });
    const c = await peekInvitation(clientJwt, { code: codes.revoked });
    expect(JSON.stringify(a)).toBe(JSON.stringify(b));
    expect(JSON.stringify(b)).toBe(JSON.stringify(c));
  });
});

function buildTestMiddleware() {
  const ipBucket = new Ratelimit({ redis, limiter: Ratelimit.slidingWindow(5, '15 m'), prefix: `${TEST_PREFIX}:ip` });
  const userBucket = new Ratelimit({ redis, limiter: Ratelimit.slidingWindow(10, '60 m'), prefix: `${TEST_PREFIX}:user` });
  return async function rl(c: any, next: any) {
    const ip = c.req.header('x-real-ip') ?? 'test-ip';
    const ipRes = await ipBucket.limit(ip);
    if (!ipRes.success) {
      c.header('Retry-After', String(Math.max(1, Math.ceil((ipRes.reset - Date.now()) / 1000))));

      return c.json(PREVIEW_ENVELOPE, 429);
    }
    return next();
  };

function percentile(sorted: number[], p: number) {
  const idx = Math.floor((p / 100) * (sorted.length - 1));
  return sorted[idx];
}

async function measure(code: string): Promise<number[]> {
  const samples: number[] = [];
  for (let i = 0; i < N_SAMPLES + N_WARMUP; i++) {
    const t0 = performance.now();
    await client.client.rpc('peek_invitation', { code_input: code });
    samples.push(performance.now() - t0);
  }
  return samples.slice(N_WARMUP).sort((a, b) => a - b);
}
