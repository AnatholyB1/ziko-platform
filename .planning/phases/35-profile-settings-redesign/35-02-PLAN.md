    **useQuery for measurements** (gallery photos):
    ```typescript
    const { data: measurementsData } = useQuery({
      queryKey: ['measurements', userId],
      queryFn: async () => {
        const { data } = await supabase
          .from('body_measurements')
          .select('id, photo_url, created_at, weight_kg')
          .eq('user_id', userId!)
          .not('photo_url', 'is', null)
          .order('created_at', { ascending: false })
          .limit(4);
        return data ?? [];
      },
      enabled: !!userId,
      staleTime: 5 * 60 * 1000,
    });
    ```

    **useQuery for badges**:
    ```typescript
    const { data: badgesData } = useQuery({
      queryKey: ['badges', userId],
      queryFn: async () => {
        // Try user_badges table first (from gamification plugin)
        const { data, error } = await supabase
          .from('user_gamification')
          .select('badges')
          .eq('user_id', userId!)
          .maybeSingle();
        if (error || !data) return [];
        // badges is a JSONB array of { id, name, icon, tint, unlocked_at }
        return (data.badges as any[]) ?? [];
      },
      enabled: !!userId,
      staleTime: 10 * 60 * 1000,
    });
    ```
