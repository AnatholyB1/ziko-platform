    export async function getProgramsForClient(jwt: string, coachId: string, clientId: string):
    - Create supabase = createUserClient(jwt)
    - Query workout_programs WHERE assigned_to_user_id = clientId AND created_by_coach_id = coachId ORDER BY start_date DESC NULLS LAST
    - For each program row, compute week_number_current: if start_date is null, return null. Otherwise: Math.floor((Date.now() - new Date(start_date).getTime()) / (7 * 24 * 60 * 60 * 1000)) + 1. Cap at weeks_count (do not exceed total).
    - For the most recent non-null start_date program, compute compliance_pct: query COUNT of workout_sessions WHERE source_program_id = program.id AND created_at >= start of current ISO week (Monday 00:00 UTC). Denominator = number of sessions in weeks_data for current week (find week matching week_number_current, count sessions array length). compliance_pct = denominator > 0 ? Math.round((done / denominator) * 100) : null.
    - Return { programs: rows with added fields week_number_current, compliance_pct (only on the first/active row; null on historical) }

    export async function upsertSharedNote(jwt: string, coachId: string, clientId: string, note: string):
    - Create supabase = createUserClient(jwt)
    - UPDATE coach_client_links SET shared_note = note WHERE coach_id = coachId AND client_id = clientId AND status = 'active'
    - Throw error if no rows updated (means the coach↔client link does not exist or is inactive — IDOR prevention)
    - Return { updated: true }

    Do NOT modify any existing function. Do NOT move or refactor the createUserClient function.
