    'use client' directive. Import createClientSupabase from '@/lib/supabase/client'. Import IoPersonOutline, IoCloudUploadOutline from 'react-icons/io5'. Import gsap from 'gsap'.

    Props interface LogoUploadProps: userId: string; currentPath: string | null; onUploaded: (bucketPath: string) => void; onRemoved: () => void.

    Constants: ALLOWED_TYPES = ['image/png', 'image/svg+xml']; MAX_BYTES = 2 * 1024 * 1024.

    State: uploading: boolean, error: string, localPreviewUrl: string | null.

    For display URL: if localPreviewUrl is set use it; else if currentPath is set construct via createClientSupabase().storage.from('coach-logos').getPublicUrl(currentPath).data.publicUrl. Compute this as displayUrl (not stored in state — derived on each render from currentPath prop).

    handleFile(file: File):
    1. setError('').
    2. Validate type: if (!ALLOWED_TYPES.includes(file.type)) { setError('Format non accepté. PNG ou SVG uniquement.'); apply GSAP shake to '.logo-upload-error'; return. }
    3. Validate size: if (file.size > MAX_BYTES) { setError('Fichier trop volumineux (max 2 Mo).'); apply GSAP shake; return. }
    4. setLocalPreviewUrl(URL.createObjectURL(file)).
    5. setUploading(true).
    6. Try: const ext = file.type === 'image/svg+xml' ? 'svg' : 'png'. const path = `${userId}/logo.${ext}`. const supabase = createClientSupabase(). const { error: uploadError } = await supabase.storage.from('coach-logos').upload(path, file, { upsert: true }). If uploadError throw it. onUploaded(path). Fire GSAP: gsap.from('.logo-avatar-img', { scale: 0.85, opacity: 0, duration: 0.25, ease: 'back.out(1.4)' }).
    7. Catch: setError('Échec du transfert. Réessayez.'); setLocalPreviewUrl(null); apply GSAP shake to '.logo-upload-error'.
    8. Finally: setUploading(false).

    GSAP shake helper (inline): gsap.to('.logo-upload-error', { x: [-4, 4, -3, 3, 0], duration: 0.3, ease: 'none' }).
