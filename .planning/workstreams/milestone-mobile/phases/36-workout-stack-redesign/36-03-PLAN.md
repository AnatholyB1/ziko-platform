TanStack Query for exercises (from research Q3 verified pattern):
```typescript
import { useQuery } from '@tanstack/react-query';
const { data: exercise, isLoading, refetch } = useQuery({
  queryKey: ['exercise', exerciseId],
  queryFn: async () => {
    const { data } = await supabase.from('exercises').select('*').eq('id', exerciseId).single();
    return data;
  },
  enabled: !!exerciseId,
});

const { data: history } = useQuery({
  queryKey: ['exercise-history', exerciseId, userId],
  queryFn: async () => {
    const { data } = await supabase
      .from('session_sets')
      .select('*, workout_sessions!inner(started_at, user_id)')
      .eq('exercise_id', exerciseId)
      .eq('workout_sessions.user_id', userId)
      .eq('completed', true)
      .order('workout_sessions.started_at', { ascending: false })
      .limit(30);
    return data ?? [];
  },
  enabled: !!exerciseId && !!userId,
});
```

| T-36-03-02 | Information Disclosure | exercises table | accept | exercises is public catalog — no sensitive data; SELECT-all is intentional |
| T-36-03-03 | Tampering | exerciseId URL param | mitigate | useQuery with enabled: !!exerciseId guards against empty param; Supabase returns 404 for non-existent IDs |
