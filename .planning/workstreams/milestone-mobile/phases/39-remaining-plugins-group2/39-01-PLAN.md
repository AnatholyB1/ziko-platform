```typescript
function MiniBars({ data, color, max }: { data: Array<{ l: string; v: number; today?: boolean }>; color: string; max?: number }) {
  const m = max ?? Math.max(...data.map((d) => d.v), 1);
  return (
    <View style={{ flexDirection: 'row', alignItems: 'flex-end', gap: 6, height: 80 }}>
      {data.map((d, i) => (
        <View key={i} style={{ flex: 1, alignItems: 'center', justifyContent: 'flex-end', height: '100%', gap: 5 }}>
          <View style={{ width: '100%', height: Math.max((d.v / m) * 80, 4), backgroundColor: d.today ? color : color + '47', borderRadius: 5 }} />
          <Text style={{ fontSize: 9.5, fontWeight: '700', color: d.today ? color : '#6B6963' }}>{d.l}</Text>
        </View>
      ))}
    </View>
  );
}
```
