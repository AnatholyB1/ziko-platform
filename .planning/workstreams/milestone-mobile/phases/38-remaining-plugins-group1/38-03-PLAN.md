function MiniBars({ data, color, max }: { data: Array<{ l: string; v: number; today?: boolean }>; color: string; max?: number }) {
  const m = max ?? Math.max(...data.map((d) => d.v), 1);
  return (
    <View style={{ flexDirection: 'row', alignItems: 'flex-end', gap: 6, height: 80 }}>
      {data.map((d, i) => (
        <View key={i} style={{ flex: 1, alignItems: 'center', justifyContent: 'flex-end', height: 80, gap: 5 }}>
          <View style={{ width: '100%', height: Math.max((d.v / m) * 72, 4), backgroundColor: d.today ? color : color + '47', borderRadius: 5 }} />
          <Text style={{ fontSize: 9.5, fontWeight: '700', color: d.today ? color : '#6B6963' }}>{d.l}</Text>
        </View>
      ))}
    </View>
  );
}

Timer countdown (D-08, RESEARCH.md Q8) :
```typescript
const intervalRef = useRef<ReturnType<typeof setInterval> | null>(null);
const startedAtRef = useRef<number | null>(null);

// Sur start :
startedAtRef.current = Date.now();
intervalRef.current = setInterval(() => {
  setSecondsLeft(s => {
    if (s <= 1) { clearInterval(intervalRef.current!); /* completion */ return 0; }
    return s - 1;
  });
}, 1000);

// AppState correction :
const sub = AppState.addEventListener('change', (nextState) => {
  if (nextState === 'active' && startedAtRef.current && isRunning) {
    const elapsed = Math.round((Date.now() - startedAtRef.current) / 1000);
    setSecondsLeft(s => Math.max(0, totalSeconds - elapsed));
  }
});
return () => { clearInterval(intervalRef.current!); sub.remove(); };
```
