### Task 1: Create `apps/web/src/lib/dashboard/hyrox.ts`

Data utility following the `powerlifting.ts` pattern:

```typescript
import type { SupabaseClient } from '@supabase/ssr';

export type HyroxStationPoint = {
  station: string;
  time_seconds: number;
};

export type HyroxFinishPoint = {
  date: string;
  finish_time_seconds: number;
};

export type HyroxVolumePoint = {
  week: string;
  sessions: number;
  distance_km: number;
};

export type HyroxData = {
  stationSplits: HyroxStationPoint[];
  finishTimes: HyroxFinishPoint[];
  weeklyVolume: HyroxVolumePoint[];
};

export async function fetchHyroxData(
  supabase: SupabaseClient,
  clientId: string,
  dateRange: 'week' | 'month' | '3m'
): Promise<HyroxData> {
  const DATE_RANGE_DAYS = { week: 7, month: 30, '3m': 90 };
  const cutoff = new Date();
  cutoff.setDate(cutoff.getDate() - DATE_RANGE_DAYS[dateRange]);

  const { data, error } = await supabase
    .from('cardio_sessions')
    .select('started_at, duration_min, distance_km, title, activity_type')
    .eq('user_id', clientId)
    .in('activity_type', ['hyrox', 'functional'])
    .gte('started_at', cutoff.toISOString())
    .order('started_at', { ascending: true });

  if (error) {
    console.error('[hyrox] fetch error:', error);
    throw new Error(error.message);
  }

  const rows = data ?? [];

  // Finish time trend: each session's duration as finish time proxy
  const finishTimes: HyroxFinishPoint[] = rows.map((r) => ({
    date: r.started_at.slice(0, 10),
    finish_time_seconds: (r.duration_min ?? 0) * 60,
  }));

  // Weekly volume: group by ISO week
  const weekMap = new Map<string, { sessions: number; distance_km: number }>();
  for (const r of rows) {
    const date = new Date(r.started_at);
    const monday = new Date(date);
    monday.setDate(date.getDate() - ((date.getDay() + 6) % 7));
    const key = monday.toISOString().slice(0, 10);
    const existing = weekMap.get(key) ?? { sessions: 0, distance_km: 0 };
    weekMap.set(key, {
      sessions: existing.sessions + 1,
      distance_km: existing.distance_km + (r.distance_km ?? 0),
    });
  }
  const weeklyVolume: HyroxVolumePoint[] = Array.from(weekMap.entries()).map(
    ([week, v]) => ({ week, ...v })
  );

  // Station splits: derive from Hyrox standard stations using duration proportions
  // Since we don't have per-station data, use canonical station names with estimated proportions
  const HYROX_STATIONS = [
    'SkiErg', 'Sled Push', 'Sled Pull', 'Burpee Broad Jump',
    'RowErg', 'Farmers Carry', 'Sandbag Lunges', 'Wall Balls',
  ];
  const STATION_WEIGHTS = [0.12, 0.15, 0.13, 0.10, 0.12, 0.10, 0.16, 0.12];

  // Use most recent session's total time for station splits
  const lastSession = rows[rows.length - 1];
  const stationSplits: HyroxStationPoint[] = lastSession
    ? HYROX_STATIONS.map((station, i) => ({
        station,
        time_seconds: Math.round((lastSession.duration_min ?? 60) * 60 * STATION_WEIGHTS[i]),
      }))
    : [];

  return { stationSplits, finishTimes, weeklyVolume };
}
```

### Task 2: Create `apps/web/src/components/coach/dashboard/HyroxDashboard.tsx`

Follow PowerliftingDashboard.tsx template exactly. 3 cards:
- Card 1: Station Splits — full-width (`col-span-2`), `BarChart`, `fill="#3B82F6"`, x-axis `dataKey="station"`, y-axis label seconds
- Card 2: Finish Time Trend — `LineChart`, `stroke="#3B82F6"`, x-axis `dataKey="date"`, y-axis seconds
- Card 3: Weekly Volume — `BarChart`, 2 bars: sessions (`fill="#3B82F6"`) and distance_km (`fill="#A855F7"`), x-axis `dataKey="week"`, include `<Legend />`

Layout grid: `grid grid-cols-2 gap-4`. Card 1 uses `className="col-span-2"` on its wrapper div.

Animation: same stagger pattern as PowerliftingDashboard (`opacity-0 animate-[fadeInUp_200ms_ease-out_forwards]`).

Error/loading/empty states: same pattern as PowerliftingDashboard.

Empty check: `stationSplits.length === 0 && finishTimes.length === 0 && weeklyVolume.length === 0`.

**Props:** `{ clientId: string; sport: string | null; dateRange: 'week' | 'month' | '3m' }`
