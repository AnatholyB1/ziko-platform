    UTILITY (exported):
    export function estimate1RM(weight: number, reps: number): number
    - if (reps === 1) return weight
    - return weight * (1 + reps / 30)

    HELPER (internal):
    function matchesLift(name: string, lift: 'squat' | 'bench' | 'deadlift'): boolean
    - const lower = name.toLowerCase()
    - return SBD_KEYWORDS[lift].some(kw => lower.includes(kw))

    function dateLabel(isoString: string): string
    - Returns YYYY-MM-DD slice: isoString.slice(0, 10)

    TRANSFORM FUNCTIONS (exported for testing):
    export function buildSBDData(rows: RawRow[]): SBDDataPoint[]
    - For each row, filter to only rows where matchesLift is true for squat/bench/deadlift
    - Group by dateLabel(row.workout_sessions.started_at)
    - For each date+lift: track max estimate1RM(weight_kg, reps)
    - Return sorted by date ascending, each entry is SBDDataPoint

    export function buildRPEData(rows: RawRow[]): RPEDataPoint[]
    - Filter out rows where rpe is null
    - Group by dateLabel(started_at)
    - For each date: avgRpe = sum of rpe / count; round to 1 decimal
    - Return sorted by date ascending

    export function buildTonnageData(rows: RawRow[]): TonnageDataPoint[]
    - For each row: compute ISO-week key (Monday-based: new Date; adjust to Monday; key = monday.toISOString().slice(0,10))
    - Group by weekKey; sum weight_kg * reps
    - Sort by weekKey ascending
    - Map to { week: 'Sem. 1', 'Sem. 2', ... (sequential index), tonnage }

    export function buildIntensityData(rows: RawRow[]): IntensityDataPoint[]
    - Group by dateLabel(started_at)
    - For each date: compute avgWeight = mean(weight_kg), avgReps = mean(reps) (round to nearest int)
    - intensity = (avgWeight / estimate1RM(avgWeight, Math.round(avgReps))) * 100; round to 1 decimal
    - Return sorted by date ascending

    MAIN FETCH FUNCTION (exported):
    export async function fetchPowerliftingData(supabase: SupabaseClient, clientId: string, dateRange: 'week' | 'month' | '3m'): Promise<PowerliftingData>
    - Compute cutoffDate using DATE_RANGE_DAYS[dateRange]
    - Execute single Supabase query with embedded joins: workout_sessions!inner and exercises!inner
    - Use .eq('workout_sessions.user_id', clientId) — clientId is the client's UUID from URL params
    - Use .gte('workout_sessions.started_at', cutoffDate.toISOString()) — NEVER session_date
    - Use .not('weight_kg', 'is', null) and .not('reps', 'is', null)
    - If error: throw new Error(error.message)
    - Cast returned data to RawRow[]
    - Return: { sbd: buildSBDData(rows), rpe: buildRPEData(rows), tonnage: buildTonnageData(rows), intensity: buildIntensityData(rows) }

    Import SupabaseClient type: import type { SupabaseClient } from '@supabase/ssr'
