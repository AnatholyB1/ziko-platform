    --- SubmitButton.tsx ---
    Props: { onPress: () => void; isAllAnswered: boolean; isLoading: boolean; label: string }
    State: ctaScaleAnim = useRef(new Animated.Value(1)).current

    Press animation:
    - handlePressIn: Animated.timing(ctaScaleAnim, { toValue: 0.97, duration: 100, useNativeDriver: true }).start()
    - handlePressOut: Animated.spring(ctaScaleAnim, { toValue: 1, tension: 160, friction: 8, useNativeDriver: true }).start()
