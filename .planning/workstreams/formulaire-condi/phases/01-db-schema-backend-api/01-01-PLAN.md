    Header comment block listing all tables and policies added.

    SET LOCAL lock_timeout = '5s';

    TABLE public.coach_forms:
      - id UUID PRIMARY KEY DEFAULT gen_random_uuid()
      - coach_id UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE
      - title TEXT NOT NULL CHECK (char_length(title) <= 200)
      - questions JSONB NOT NULL DEFAULT '[]'::jsonb   — array of question objects per D-02: {id, type, label, options?}
      - trigger_config JSONB NOT NULL DEFAULT '{"type":"manual"}'::jsonb   — per D-06
      - status TEXT NOT NULL DEFAULT 'draft' CHECK (status IN ('draft','active','archived'))
      - created_at TIMESTAMPTZ NOT NULL DEFAULT now()
      - updated_at TIMESTAMPTZ NOT NULL DEFAULT now()
    ENABLE ROW LEVEL SECURITY.
    POLICY "coach_forms_own" FOR ALL USING (auth.uid() = coach_id) WITH CHECK (auth.uid() = coach_id).

    TABLE public.form_instances:
      - id UUID PRIMARY KEY DEFAULT gen_random_uuid()
      - form_id UUID NOT NULL REFERENCES public.coach_forms(id) ON DELETE CASCADE
      - athlete_id UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE
      - status TEXT NOT NULL DEFAULT 'pending' CHECK (status IN ('pending','submitted'))
      - created_at TIMESTAMPTZ NOT NULL DEFAULT now()
      - submitted_at TIMESTAMPTZ NULL
    ENABLE ROW LEVEL SECURITY.
    POLICY "form_instances_coach" FOR SELECT USING (public.is_coach_of(auth.uid(), athlete_id) OR EXISTS (SELECT 1 FROM public.coach_forms cf WHERE cf.id = form_id AND cf.coach_id = auth.uid())).
    POLICY "form_instances_athlete" FOR SELECT USING (auth.uid() = athlete_id).
    POLICY "form_instances_insert" FOR INSERT WITH CHECK (EXISTS (SELECT 1 FROM public.coach_forms cf WHERE cf.id = form_id AND cf.coach_id = auth.uid()) OR public.is_coach_of((SELECT cf.coach_id FROM public.coach_forms cf WHERE cf.id = form_id), athlete_id)).
    POLICY "form_instances_athlete_update" FOR UPDATE USING (auth.uid() = athlete_id) WITH CHECK (auth.uid() = athlete_id).

    TABLE public.form_responses:
      - id UUID PRIMARY KEY DEFAULT gen_random_uuid()
      - instance_id UUID NOT NULL REFERENCES public.form_instances(id) ON DELETE CASCADE
      - athlete_id UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE
      - answers JSONB NOT NULL DEFAULT '[]'::jsonb   — per D-04: [{question_id, value}] where value is string|number|boolean
      - submitted_at TIMESTAMPTZ NOT NULL DEFAULT now()
    ENABLE ROW LEVEL SECURITY.
    POLICY "form_responses_coach_read" FOR SELECT USING (public.is_coach_of(auth.uid(), athlete_id)).
    POLICY "form_responses_athlete" USING (auth.uid() = athlete_id) WITH CHECK (auth.uid() = athlete_id).

    UNIQUE INDEX form_instances_no_dup_pending per D-09:
      CREATE UNIQUE INDEX form_instances_no_dup_pending ON public.form_instances(form_id, athlete_id) WHERE status = 'pending';

    INDEX idx_form_instances_athlete ON public.form_instances(athlete_id) — for mobile overlay queries (Claude's discretion).
    INDEX idx_form_instances_form ON public.form_instances(form_id) — for trigger engine queries.
    INDEX idx_form_responses_instance ON public.form_responses(instance_id).

    End-of-migration comment.

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Supabase RLS | Auth-validated JWT → row-level access gating per coach_id / athlete_id |
| Migration execution | SQL runs with superuser privileges; a mistake here is not reversible without a new migration |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-01-01 | Elevation of Privilege | form_instances RLS — athlete reads another athlete's instances | mitigate | RLS policy "form_instances_athlete" enforces auth.uid() = athlete_id |
| T-01-02 | Information Disclosure | coach_forms readable by athletes via direct Supabase client | mitigate | Per D-07: no athlete RLS SELECT on coach_forms; questions exposed only through API join endpoint in plan 01-02 |
| T-01-03 | Tampering | Duplicate pending instances created by racing INSERTs | mitigate | Partial UNIQUE INDEX form_instances_no_dup_pending enforces uniqueness at DB level — race condition safe |
| T-01-04 | Denial of Service | Unbounded JSONB growth for questions or answers columns | accept | v1.14 scope; forms are coach-authored and short-lived assessment data; no PII at risk |
| T-01-SC | Tampering | supabase db push runs migration against wrong project | mitigate | Executor verifies SUPABASE_URL env points to correct project before pushing |
</threat_model>
