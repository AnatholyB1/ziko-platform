    2. Functions: `SELECT p.proname, p.prosecdef AS security_definer, pg_get_function_identity_arguments(p.oid) AS args FROM pg_proc p JOIN pg_namespace n ON n.oid = p.pronamespace WHERE n.nspname = 'public' ORDER BY p.proname;`

    5. Storage buckets + usage: `SELECT b.id, b.name, b.public, count(o.id) AS object_count, pg_size_pretty(coalesce(sum((o.metadata->>'size')::bigint),0)) AS total_size FROM storage.buckets b LEFT JOIN storage.objects o ON o.bucket_id = b.id GROUP BY b.id, b.name, b.public ORDER BY b.name;`
    6. Storage RLS policies: `SELECT policyname, cmd, qual, with_check FROM pg_policies WHERE schemaname = 'storage' AND tablename = 'objects' ORDER BY policyname;`
    7. Extensions: `SELECT extname, extversion FROM pg_extension ORDER BY extname;`
    8. Realtime publication membership: `SELECT tablename FROM pg_publication_tables WHERE pubname = 'supabase_realtime' ORDER BY tablename;`
    9. Non-UUID sequence-backed PKs: `SELECT table_name, column_name, column_default FROM information_schema.columns WHERE table_schema = 'public' AND column_default LIKE 'nextval%';`
    10. DB size: `SELECT pg_size_pretty(pg_database_size(current_database())) AS db_size;`
    11. Connections: `SHOW max_connections;`
    12. Auth user count (non-PII — count only, never select `email`): `SELECT count(*) FROM auth.users;`
